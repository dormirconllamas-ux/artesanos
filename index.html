Arquitectura & Backend

Código Next.js + Supabase + Vercel

Estructura completa para la implementación real fullstack de esta propuesta con Supabase DB y API Routes de Next.js.

1. Esquema SQL para Supabase (Database Schema) supabase_schema.sql 
-- Tabla Vendedores
CREATE TABLE vendors (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  category TEXT CHECK (category IN ('Artesano', 'Agricultor')),
  station TEXT NOT NULL,
  whatsapp TEXT NOT NULL,
  is_available BOOLEAN DEFAULT true,
  bio TEXT,
  avatar_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);

-- Tabla Productos
CREATE TABLE products (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  vendor_id UUID REFERENCES vendors(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  price NUMERIC NOT NULL,
  description TEXT,
  image_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now())
);
2. Next.js App Router: API Route `/app/api/toggle-stock/route.ts` Next.js 14+ 
import { createClient } from '@supabase/supabase-js';
import { NextResponse } from 'next/server';

const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

export async function POST(request: Request) {
  try {
    const { vendorId, isAvailable } = await request.json();

    const { data, error } = await supabase
      .from('vendors')
      .update({ is_available: isAvailable })
      .eq('id', vendorId)
      .select();

    if (error) throw error;

    return NextResponse.json({ success: true, vendor: data[0] });
  } catch (error: any) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}
